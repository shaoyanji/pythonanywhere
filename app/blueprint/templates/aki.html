This website runs a weird interactivity script called aki. it's essentially controls on html navigation using json.
source code: 

         window.onload=()=>{
            document.querySelectorAll("[url][fn]").forEach(e=>{
                const r=e.attributes;e[r.fn.value]=()=>{
                    fetch(r.url.value).then(e=>e.json()).then(e=>{
                        Object.entries(e).forEach(e=>{
                            Object.entries(e[1]).forEach(r=>{
                                document.querySelector(e[0])[r[0]]=r[1]
                            })
                        })
                    })
                }
            })
        };
